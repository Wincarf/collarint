   205→   205→
   206→   206→function PlanTrail({ plan, sessions }: { plan: MentorshipDTO["plan"]; sessions: SessionDTO[] }) {
   207→   207→  if (!plan) return null;
   208→   208→  const completed = sessions.filter((s) => s.completed).length;
   209→   209→
   210→   210→  return (
   211→   211→    <Card className="border-border shadow-sm">
   212→   212→      <CardContent className="p-5 sm:p-6">
   213→   213→        <SectionTitle
   214→   214→          title="Plano de mentoria"
   215→   215→          subtitle={
   216→   216→            plan.generatedBy === "ai"
   217→   217→              ? "Trilha gerada por IA a partir do seu objetivo"
   218→   218→              : "Trilha estruturada em 4 sessões"
   219→   219→          }
   220→   220→          action={
   221→   221→            <span className="rounded-full bg-gold-soft px-3 py-1 text-xs font-semibold text-[#7a5c1f]">
   222→   222→              {completed}/{plan.sessions.length} sessões
   223→   223→            </span>
   224→   224→          }
   225→   225→        />
   226→   226→        <ol className="mt-5 space-y-0">
   227→   227→          {plan.sessions.map((s, i) => {
   228→   228→            const done = i < completed;
   229→   229→            const current = i === completed;
   230→   230→            return (
   231→   231→              <li key={s.number} className="relative flex gap-4 pb-6 last:pb-0">
   232→   232→                {i < plan.sessions.length - 1 && (
   233→   233→                  <span
   234→   234→                    className={cn("absolute left-[15px] top-8 h-full w-0.5", done ? "bg-gold" : "bg-border")}
   235→   235→                    aria-hidden="true"
   236→   236→                  />
   237→   237→                )}
   238→   238→                <span
   239→   239→                  className={cn(
   240→   240→                    "relative z-10 flex h-8 w-8 shrink-0 items-center justify-center rounded-full border-2 text-xs font-bold",
   241→   241→                    done
   242→   242→                      ? "border-gold bg-gold text-navy"
   243→   243→                      : current
   244→   244→                        ? "border-gold bg-card text-[#7a5c1f]"
   245→   245→                        : "border-border bg-card text-muted-foreground"
   246→   246→                  )}
   247→   247→                >
   248→   248→                  {done ? <Check className="h-4 w-4" /> : s.number}
   249→   249→                </span>
   250→   250→                <div className="min-w-0 pt-1">
   251→   251→                  <p className={cn("font-semibold", done || current ? "text-navy" : "text-muted-foreground")}>
   252→   252→                    {s.title}
   253→   253→                  </p>
   254→   254→                  <p className="mt-0.5 text-sm text-muted-foreground">{s.objective}</p>
   255→   255→                  {s.topics.length > 0 && (
   256→   256→                    <div className="mt-2 flex flex-wrap gap-1.5">
   257→   257→                      {s.topics.map((t) => (
   258→   258→                        <span key={t} className="rounded-md bg-secondary px-2 py-0.5 text-xs text-muted-foreground">
   259→   259→                          {t}
   260→   260→                        </span>
   261→   261→                      ))}
   262→   262→                    </div>
   263→   263→                  )}
   264→   264→                </div>
   265→   265→              </li>
   266→   266→            );
   267→   267→          })}
   268→   268→        </ol>
   269→   269→      </CardContent>
   270→   270→    </Card>
   271→   271→  );
   272→   272→}

   276→   276→function SessionsSection({
   277→   277→  mentorship,
   278→   278→  role,
   279→   279→  onChanged,
   280→   280→}: {
   281→   281→  mentorship: MentorshipDTO;
   282→   282→  role: "mentee" | "mentor";
   283→   283→  onChanged: () => void;
   284→   284→}) {
   285→   285→  const [recording, setRecording] = useState<SessionDTO | null>(null);
   286→   286→  const [notes, setNotes] = useState("");
   287→   287→  const [commitments, setCommitments] = useState("");
   288→   288→  const [saving, setSaving] = useState(false);
   289→   289→  const { toast } = useToast();
   290→   290→
   291→   291→  const upcoming = mentorship.sessions.filter((s) => !s.completed);
   292→   292→  const past = mentorship.sessions.filter((s) => s.completed);
   293→   293→
   294→   294→  async function saveRecord(complete: boolean) {
   295→   295→    if (!recording) return;
   296→   296→    if (complete && notes.trim().length === 0) {
   297→   297→      toast({ title: "Descreva o que foi discutido", variant: "destructive" });
   298→   298→      return;
   299→   299→    }
   300→   300→    setSaving(true);
   301→   301→    try {
   302→   302→      await api.recordSession(mentorship.id, recording.id, notes.trim(), commitments.trim(), complete);
   303→   303→      toast({ title: complete ? "Sessão registrada!" : "Registro salvo." });
   304→   304→      setRecording(null);
   305→   305→      setNotes("");
   306→   306→      setCommitments("");
   307→   307→      onChanged();
   308→   308→    } catch (e) {
   309→   309→      toast({ title: "Erro ao registrar", description: e instanceof Error ? e.message : undefined, variant: "destructive" });
   310→   310→    } finally {
   311→   311→      setSaving(false);
   312→   312→    }
   313→   313→  }
   314→   314→
   315→   315→  return (
   316→   316→    <Card className="border-border shadow-sm">
   317→   317→      <CardContent className="p-5 sm:p-6">
   318→   318→        <SectionTitle title="Sessões" subtitle="Agendamentos, registros e compromissos" />
   319→   319→        <div className="mt-4 space-y-3">
   320→   320→          {upcoming.length === 0 && past.length === 0 && (
   321→   321→            <p className="rounded-lg border border-dashed border-border p-6 text-center text-sm text-muted-foreground">
   322→   322→              Nenhuma sessão ainda.
   323→   323→              {role === "mentor" ? " Agende a primeira sessão." : " Seu mentor vai agendar a primeira sessão."}
   324→   324→            </p>
   325→   325→          )}
   326→   326→
   327→   327→          {upcoming.map((s) => (
   328→   328→            <div key={s.id} className="flex flex-wrap items-center justify-between gap-3 rounded-lg border border-gold/40 bg-gold-soft/40 p-4">
   329→   329→              <div className="flex items-center gap-3">
   330→   330→                <span className="flex h-9 w-9 items-center justify-center rounded-full bg-gold text-navy">
   331→   331→                  <CalendarPlus className="h-4 w-4" />
   332→   332→                </span>
   333→   333→                <div>
   334→   334→                  <p className="text-sm font-semibold text-navy">
   335→   335→                    {s.scheduledAt
   336→   336→                      ? format(new Date(s.scheduledAt), "EEEE, d 'de' MMMM 'às' HH:mm", { locale: ptBR })
   337→   337→                      : "Data a definir"}
   338→   338→                  </p>
   339→   339→                  <p className="text-xs text-muted-foreground">Próxima sessão agendada</p>
   340→   340→                </div>
   341→   341→              </div>
   342→   342→              {role === "mentor" && (
   343→   343→                <Button
   344→   344→                  size="sm"
   345→   345→                  variant="outline"
   346→   346→                  onClick={() => {
   347→   347→                    setRecording(s);
   348→   348→                    setNotes("");
   349→   349→                    setCommitments("");
   350→   350→                  }}
   351→   351→                  className="gap-1 border-gold/50"
   352→   352→                >
   353→   353→                  <ClipboardList className="h-4 w-4" /> Registrar sessão
   354→   354→                </Button>
   355→   355→              )}
   356→   356→            </div>
   357→   357→          ))}
   358→   358→
   359→   359→          {[...past].reverse().map((s) => (
   360→   360→            <div key={s.id} className="rounded-lg border border-border bg-card p-4">
   361→   361→              <div className="flex flex-wrap items-center justify-between gap-2">
   362→   362→                <p className="text-sm font-semibold text-navy">
   363→   363→                  {s.scheduledAt
   364→   364→                    ? format(new Date(s.scheduledAt), "d 'de' MMMM 'de' yyyy 'às' HH:mm", { locale: ptBR })
   365→   365→                    : "Sessão"}
   366→   366→                </p>
   367→   367→                {role === "mentor" && (
   368→   368→                  <Button
   369→   369→                    size="sm"
   370→   370→                    variant="ghost"
   371→   371→                    className="h-7 gap-1 text-xs text-muted-foreground"
   372→   372→                    onClick={() => {
   373→   373→                      setRecording(s);
   374→   374→                      setNotes(s.notes ?? "");
   375→   375→                      setCommitments(s.commitments ?? "");
   376→   376→                    }}
   377→   377→                  >
   378→   378→                    Editar registro
   379→   379→                  </Button>
   380→   380→                )}
   381→   381→              </div>
   382→   382→              {s.notes && (
   383→   383→                <div className="mt-2 text-sm">
   384→   384→                  <span className="font-medium text-navy">O que foi discutido: </span>
   385→   385→                  <span className="text-muted-foreground">{s.notes}</span>
   386→   386→                </div>
   387→   387→              )}
   388→   388→              {s.commitments && (
   389→   389→                <div className="mt-1.5 text-sm">
   390→   390→                  <span className="font-medium text-navy">Compromissos assumidos: </span>
   391→   391→                  <span className="text-muted-foreground">{s.commitments}</span>
   392→   392→                </div>
   393→   393→              )}
   394→   394→              {!s.notes && !s.commitments && <p className="mt-1 text-xs text-muted-foreground">Sem registro detalhado.</p>}
   395→   395→            </div>
   396→   396→          ))}
   397→   397→        </div>
   398→   398→      </CardContent>
   399→   399→
   400→   400→      <Dialog open={!!recording} onOpenChange={(open) => !open && setRecording(null)}>
   401→   401→        <DialogContent className="sm:max-w-lg">
   402→   402→          <DialogHeader>
   403→   403→            <DialogTitle className="text-navy">Registro da sessão</DialogTitle>
   404→   404→          </DialogHeader>
   405→   405→          <div className="space-y-4">
   406→   406→            <div className="space-y-2">
   407→   407→              <Label htmlFor="sess-notes">O que foi discutido</Label>
   408→   408→              <Textarea
   409→   409→                id="sess-notes"
   410→   410→                rows={4}
   411→   411→                placeholder="Ex.: Mapeamos o funil de vendas e identificamos que o diagnóstico do decisor é o ponto fraco..."
   412→   412→                value={notes}
   413→   413→                onChange={(e) => setNotes(e.target.value)}
   414→   414→              />
   415→   415→            </div>
   416→   416→            <div className="space-y-2">
   417→   417→              <Label htmlFor="sess-commit">Compromissos assumidos</Label>
   418→   418→              <Textarea
   419→   419→                id="sess-commit"
   420→   420→                rows={3}
   421→   421→                placeholder="Ex.: Mentorado vai mapear os 3 decisores; mentor vai enviar o template de research."
   422→   422→                value={commitments}
   423→   423→                onChange={(e) => setCommitments(e.target.value)}
   424→   424→              />
   425→   425→            </div>
   426→   426→          </div>
   427→   427→          <DialogFooter className="gap-2 sm:gap-0">
   428→   428→            <Button variant="outline" onClick={() => saveRecord(false)} disabled={saving}>
   429→   429→              Salvar rascunho
   430→   430→            </Button>
   431→   431→            <Button
   432→   432→              onClick={() => saveRecord(true)}
   433→   433→              disabled={saving}
   434→   434→              className="bg-navy hover:bg-navy-light gap-2"
   435→   435→            >
   436→   436→              {saving ? <Loader2 className="h-4 w-4 animate-spin" /> : <CheckCircle2 className="h-4 w-4" />}
   437→   437→              Concluir sessão
   438→   438→            </Button>
   439→   439→          </DialogFooter>
   440→   440→        </DialogContent>
   441→   441→      </Dialog>
   442→   442→    </Card>
   443→   443→  );
   444→   444→}

   463→   463→  const pending = tasks.filter((t) => !t.completed);
   464→   464→  const done = tasks.filter((t) => t.completed);
   465→   465→
   466→   466→  async function toggle(task: TaskDTO) {
   467→   467→    try {
   468→   468→      await api.toggleTask(task.id, !task.completed);
   469→   469→      onChanged();
   470→   470→    } catch (e) {
   471→   471→      toast({ title: "Erro", description: e instanceof Error ? e.message : undefined, variant: "destructive" });
   472→   472→    }
   473→   473→  }
   474→   474→
   475→   475→  async function add() {
   476→   476→    if (newTitle.trim().length < 3) return;
   477→   477→    setAdding(true);
   478→   478→    try {
   479→   479→      await api.createTask(mentorshipId, newTitle.trim());
   480→   480→      setNewTitle("");
   481→   481→      onChanged();
   482→   482→    } catch (e) {
   483→   483→      toast({ title: "Erro", description: e instanceof Error ? e.message : undefined, variant: "destructive" });
   484→   484→    } finally {
   485→   485→      setAdding(false);
   486→   486→    }
   487→   487→  }
   488→   488→
   489→   489→  return (
   490→   490→    <Card className="border-border shadow-sm">
   491→   491→      <CardContent className="p-5">
   492→   492→        <SectionTitle title="Tarefas" subtitle={editable ? "Criadas por você e pelo Coach" : "Acompanhamento do mentorado"} />

   495→   495→            <p className="py-4 text-center text-sm text-muted-foreground">
   496→   496→              Nenhuma tarefa ainda. {editable ? "O Coach cria tarefas no chat — ou adicione a primeira." : ""}

   498→   498→          )}
   499→   499→          {pending.map((t) => (
   500→   500→            <TaskRow key={t.id} task={t} onToggle={toggle} />

   502→   502→          {done.map((t) => (
   503→   503→            <TaskRow key={t.id} task={t} onToggle={toggle} />
   504→   504→          ))}
   505→   505→        </div>
   506→   506→        {editable && (
   507→   507→          <div className="mt-4 flex gap-2">
   508→   508→            <Input
   509→   509→              placeholder="Nova tarefa..."

   517→   517→          </div>

   556→   556→function CoachChat({
   557→   557→  mentorshipId,
   558→   558→  mentorName,
   559→   559→  planReady,
   560→   560→  onTaskCreated,
   561→   561→}: {
   562→   562→  mentorshipId: string;
   563→   563→  mentorName: string;
   564→   564→  planReady: boolean;
   565→   565→  onTaskCreated: () => void;
   566→   566→}) {
   567→   567→  const [messages, setMessages] = useState<CoachMsg[]>([]);
   568→   568→  const [loaded, setLoaded] = useState(false);
   569→   569→  const [input, setInput] = useState("");
   570→   570→  const [sending, setSending] = useState(false);
   571→   571→  const [preparing, setPreparing] = useState(false);
   572→   572→  const [prepResult, setPrepResult] = useState<{ content: string } | null>(null);
   573→   573→  const [copied, setCopied] = useState(false);
   574→   574→  const scrollRef = useRef<HTMLDivElement>(null);
   575→   575→  const { toast } = useToast();
   576→   576→
   577→   577→  useEffect(() => {
   578→   578→    api
   579→   579→      .coachHistory(mentorshipId)
   580→   580→      .then((data) => setMessages(data.messages))
   581→   581→      .catch(() => undefined)
   582→   582→      .finally(() => setLoaded(true));
   583→   583→  }, [mentorshipId]);
   584→   584→
   585→   585→  useEffect(() => {
   586→   586→    scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight, behavior: "smooth" });
   587→   587→  }, [messages, sending]);
   588→   588→
   589→   589→  async function send() {
   590→   590→    const text = input.trim();
   591→   591→    if (!text || sending) return;
   592→   592→    setInput("");
   593→   593→    setSending(true);
   594→   594→    const optimistic: CoachMsg = {
   595→   595→      id: `tmp-${Date.now()}`,
   596→   596→      role: "user",
   597→   597→      content: text,
   598→   598→      createdAt: new Date().toISOString(),
   599→   599→    };
   600→   600→    setMessages((m) => [...m, optimistic]);
   601→   601→    try {
   602→   602→      const res = await api.coachSend(mentorshipId, text);
   603→   603→      setMessages((m) => [...m.filter((x) => x.id !== optimistic.id), res.userMessage, res.assistantMessage]);
   604→   604→      if (res.createdTasks.length > 0) {
   605→   605→        toast({
   606→   606→          title: "Tarefa criada pelo Coach",
   607→   607→          description: res.createdTasks.map((t) => t.title).join(" · "),
   608→   608→        });
   609→   609→        onTaskCreated();
   610→   610→      }
   611→   611→    } catch (e) {
   612→   612→      setMessages((m) => m.filter((x) => x.id !== optimistic.id));
   613→   613→      toast({ title: "Falha ao enviar", description: e instanceof Error ? e.message : undefined, variant: "destructive" });
   614→   614→    } finally {
   615→   615→      setSending(false);
   616→   616→    }
   617→   617→  }
   618→   618→
   619→   619→  async function prepareSession() {
   620→   620→    setPreparing(true);
   621→   621→    try {
   622→   622→      const res = await api.prepare(mentorshipId);
   623→   623→      setPrepResult(res.prep);
   624→   624→      toast({ title: "Preparação pronta!", description: "Copie a pauta e leve para a sessão." });
   625→   625→      onTaskCreated();
   626→   626→    } catch (e) {
   627→   627→      toast({ title: "Erro ao preparar", description: e instanceof Error ? e.message : undefined, variant: "destructive" });
   628→   628→    } finally {
   629→   629→      setPreparing(false);
   630→   630→    }
   631→   631→  }
   632→   632→
   633→   633→  async function copyPrep() {
   634→   634→    const content = prepResult?.content;
   635→   635→    if (!content) return;
   636→   636→    try {
   637→   637→      await navigator.clipboard.writeText(content);
   638→   638→      setCopied(true);
   639→   639→      setTimeout(() => setCopied(false), 2000);
   640→   640→    } catch {
   641→   641→      toast({ title: "Não foi possível copiar automaticamente", description: "Selecione o texto e copie manualmente." });
   642→   642→    }
   643→   643→  }
   644→   644→
   645→   645→  return (
   646→   646→    <Card className="flex h-[640px] flex-col border-border shadow-sm">
   647→   647→      <div className="border-b border-border p-4">
   648→   648→        <div className="flex items-center gap-2">
   649→   649→          <span className="flex h-8 w-8 items-center justify-center rounded-full bg-navy text-gold">
   650→   650→            <Sparkles className="h-4 w-4" />
   651→   651→          </span>
   652→   652→          <div className="min-w-0 flex-1">
   653→   653→            <p className="text-sm font-bold text-navy">IA Coach</p>
   654→   654→            <p className="text-xs text-muted-foreground">Apoio entre sessões · sempre com o contexto da sua mentoria</p>
   655→   655→          </div>
   656→   656→        </div>
   657→   657→        <Button
   658→   658→          size="sm"
   659→   659→          onClick={prepareSession}
   660→   660→          disabled={preparing || !planReady}
   661→   661→          className="mt-3 w-full gap-2 bg-gold font-semibold text-navy hover:bg-gold/90"
   662→   662→        >
   663→   663→          {preparing ? <Loader2 className="h-4 w-4 animate-spin" /> : <ClipboardList className="h-4 w-4" />}
   664→   664→          Preparar próxima sessão
   665→   665→        </Button>
   666→   666→      </div>
   667→   667→
   668→   668→      <div ref={scrollRef} className="scroll-slim flex-1 space-y-3 overflow-y-auto p-4">
   669→   669→        {!loaded && (
   670→   670→          <div className="flex justify-center py-8">
   671→   671→            <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   672→   672→          </div>
   673→   673→        )}
   674→   674→        {loaded && messages.length === 0 && (
   675→   675→          <div className="rounded-lg bg-secondary p-4 text-sm text-muted-foreground">
   676→   676→            Olá! Eu sou seu Coach Collarint. Conheço seu objetivo, seu plano com {mentorName} e suas tarefas.
   677→   677→            Me pergunte o que precisar — por exemplo: <em>&ldquo;como devo me preparar para a próxima sessão?&rdquo;</em>
   678→   678→          </div>
   679→   679→        )}
   680→   680→        {messages.map((m) => (
   681→   681→          <div key={m.id} className={cn("flex", m.role === "user" ? "justify-end" : "justify-start")}>
   682→   682→            <div
   683→   683→              className={cn(
   684→   684→                "max-w-[92%] rounded-2xl px-3.5 py-2.5 text-sm",
   685→   685→                m.role === "user"
   686→   686→                  ? "rounded-br-md bg-navy text-white"
   687→   687→                  : "rounded-bl-md border border-border bg-card text-foreground"
   688→   688→              )}
   689→   689→            >
   690→   690→              {m.role === "assistant" ? (
   691→   691→                <MarkdownContent content={m.content} />
   692→   692→              ) : (
   693→   693→                <p className="whitespace-pre-wrap">{m.content}</p>
   694→   694→              )}
   695→   695→            </div>
   696→   696→          </div>
   697→   697→        ))}
   698→   698→        {sending && (
   699→   699→          <div className="flex justify-start">
   700→   700→            <div className="rounded-2xl rounded-bl-md border border-border bg-card px-4 py-3">
   701→   701→              <span className="flex gap-1">
   702→   702→                <span className="typing-dot h-2 w-2 rounded-full bg-muted-foreground" />
   703→   703→                <span className="typing-dot h-2 w-2 rounded-full bg-muted-foreground" />
   704→   704→                <span className="typing-dot h-2 w-2 rounded-full bg-muted-foreground" />
   705→   705→              </span>
   706→   706→            </div>
   707→   707→          </div>
   708→   708→        )}
   709→   709→      </div>
   710→   710→
   711→   711→      <div className="border-t border-border p-3">
   712→   712→        <form
   713→   713→          className="flex gap-2"
   714→   714→          onSubmit={(e) => {
   715→   715→            e.preventDefault();
   716→   716→            send();
   717→   717→          }}
   718→   718→        >
   719→   719→          <Input
   720→   720→            placeholder="Escreva para o Coach..."
   721→   721→            value={input}
   722→   722→            onChange={(e) => setInput(e.target.value)}
   723→   723→            disabled={sending}
   724→   724→          />
   725→   725→          <Button type="submit" size="icon" className="shrink-0 bg-navy hover:bg-navy-light" disabled={sending || !input.trim()} aria-label="Enviar mensagem">
   726→   726→            <Send className="h-4 w-4" />
   727→   727→          </Button>
   728→   728→        </form>
   729→   729→      </div>
   730→   730→
   731→   731→      <Dialog open={!!prepResult} onOpenChange={(open) => !open && setPrepResult(null)}>
   732→   732→        <DialogContent className="max-h-[80vh] sm:max-w-lg">
   733→   733→          <DialogHeader>
   734→   734→            <DialogTitle className="text-navy">Sua preparação para a próxima sessão</DialogTitle>
   735→   735→          </DialogHeader>
   736→   736→          <div className="scroll-slim max-h-[50vh] overflow-y-auto pr-2 text-sm">
   737→   737→            <MarkdownContent content={prepResult?.content ?? ""} />

   746→   746→      </Dialog>

   748→   748→  );
